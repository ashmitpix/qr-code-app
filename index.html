qrcode pillow import streamlit as st import qrcode from io import BytesIO # Hide the Streamlit menus completely st.markdown(""" """, unsafe_allow_html=True) st.title("QR Code Generator") st.write("Enter a website URL or text below to generate a downloadable QR code.") user_input = st.text_input("Website URL or Text:", placeholder="https://example.com") if st.button("Generate QR Code"): if user_input: qr = qrcode.QRCode(version=1, box_size=10, border=4) qr.add_data(user_input) qr.make(fit=True) img = qr.make_image(fill_color="black", back_color="white") buffer = BytesIO() img.save(buffer, format="PNG") buffer.seek(0) st.image(buffer, caption="Your Generated QR Code", width=250) st.download_button(label="Download PNG", data=buffer.getvalue(), file_name="qrcode.png", mime="image/png") else: st.warning("Please enter some text or a URL first!")
